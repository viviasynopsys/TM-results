# Read the parameters
angle_1=%angle 1% *pi/180;
angle_2=%angle 2% *pi/180;
holo_type=%holo type%; # R or T
n0=%n0%; # refractive index
Dn=%Dn%; #modulation index                
w_rec=%w_rec%; # recording wavelength

# Compute the K vectors
if (holo_type=='R'){direction=-1;}
else if (holo_type=='T'){direction=1;}
k1=[sin(angle_1), 0, cos(angle_1)];
k2=[sin(angle_2), 0, cos(angle_2)*direction];
K=k1-k2;

# Compute the periodicity
dx=abs(w_rec/(n0*K(1)));
dz=abs(w_rec/(n0*K(3)));

# Get the material modulation
        modulation =num2str(n0) + "+" 
                    + num2str(Dn) 
                    + "*cos((2*pi*" + num2str(n0) + "/" + num2str(w_rec*1e6) 
                    + ")*(x* (" +num2str(K(1))+")+y*("+num2str(K(2))+")+z*("+num2str(K(3))+")))" ;
        ?modulation; 
        
# Setup the system
setnamed("rectangle","index",modulation);
setnamed("rectangle","z span",dz); 
setnamed("RCWA","x span",dx);
setnamed("RCWA","z span",dz*1.5);  
setnamed("RCWA","index",n0); 
setnamed("RCWA","wavelength center",w_rec);

# Read the parameters
angle_1=%angle 1% *pi/180;
angle_2=%angle 2% *pi/180;
holo_type=%holo type%; # R or T
n0=%n0%; # refractive index
Dn=%Dn%; #modulation index                
w_rec=%w_rec%; # recording wavelength

# Compute the K vectors
if (holo_type=='R'){direction=-1;}
else if (holo_type=='T'){direction=1;}
k1=[sin(angle_1), 0, cos(angle_1)];
k2=[sin(angle_2), 0, cos(angle_2)*direction];
K=k1-k2;

# Compute the periodicity
dx=abs(w_rec/(n0*K(1)));
dz=abs(w_rec/(n0*K(3)));

# Get the material modulation
        modulation =num2str(n0) + "+" 
                    + num2str(Dn) 
                    + "*cos((2*pi*" + num2str(n0) + "/" + num2str(w_rec*1e6) 
                    + ")*(x* (" +num2str(K(1))+")+y*("+num2str(K(2))+")+z*("+num2str(K(3))+")))" ;
        ?modulation; 
        
# Setup the system
setnamed("rectangle","index",modulation);
setnamed("rectangle","z span",dz); 
setnamed("RCWA","x span",dx);
setnamed("RCWA","z span",dz*1.5);  
setnamed("RCWA","index",n0);
